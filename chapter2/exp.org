(define (accumulate op init seq)
  (if (null? seq) init
      (op (car seq) (accumulate op init (cdr seq)))))

(define (flatmap proc seq)
  (accumulate append nil (map proc seq)))

(define (filter f seq)
  (accumulate (lambda (x acc) (if (f x) (cons x acc) acc)) nil seq))

(define (enumerate-interval a b)
  (define (iter i acc)
    (if (< i a) acc
        (iter (dec i) (cons i acc))))
  (if (> a b) (enumerate-interval b a)
      (iter b nil)))

(define (prime? n)
  (define (next-odd x)
    (if (odd? x) x (dec x)))
  (define (iter i)
    (cond [(<= i 1) #t]
          [(= (modulo n i) 0) #f]
          [else (iter (- i 2))]))
  (cond [(= n 1) #f]
        [(even? n) #f]
        [else (iter (next-odd (inc (floor (sqrt n)))))]))

(define (prime-sum? pair)
  (prime? (+ (car pair) (cadr pair))))

(define (make-pair-sum pair)
  (list (car pair) (cadr pair) (+ (car pair) (cadr pair))))

(define (prime-sum-pairs n)
  (map make-pair-sum
       (filter prime-sum?
               (flatmap (lambda (i)
                          (map (lambda (j) (list i j))
                               (enumerate-interval 1 (dec i))))
                        (enumerate-interval 1 n)))))

(prime-sum-pairs 9)
