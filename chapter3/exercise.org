(define (monte-carlo trials experiment)
  (define (iter trials-remaining trials-passed)
    (cond [(zero? trials-remaining) (/ trials-passed trials)]
          [(experiment) (iter (- trials-remaining 1) (+ trials-passed 1))]
          [else (iter (- trials-remaining 1) trials-passed)]))
  (iter trials 0))
#+end_src

(define (random-f int . precisions)
  (define default-precision 1000000)
  (define (random-by-precision precision)
    (* int (/ (random precision) precision)))
  (if (null? precisions)
      (random-by-precision default-precision)
      (map random-by-precision precisions)))

(define (random-in-range low high)
  (let ([range (- high low)])
    (+ low (random-f range))))

(define (estimate-integral p x1 x2 y1 y2 trials)
  (define (random-point x1 x2 y1 y2)
    (cond [(> x1 x2) (random-point x2 x1 y1 y2)]
          [(> y1 y2) (random-point x1 x2 y2 y1)]
          [else (list (random-in-range x1 x2)
                      (random-in-range y1 y2))]))
  (define (experiment)
    (apply p (random-point x1 x2 y1 y2)))
  (monte-carlo trials experiment))

Well, the test code in the above exercises are all making use of the sequence of evaluation.

#+begin_src racket :noweb yes
(define (plus-left-right proc1 proc2)
  (+ (proc1) (proc2)))

(define (plus-right-left proc1 proc2)
  (+ (proc2) (proc1)))

(define (get-and-set x)
  (lambda (y)
    (let ([before x])

             before))))

#+begin_src racket :results none
(define (id x) x)

(define (always res) (lambda _args res))

(define (identity . args) args)

(define (compose f g)
  (lambda args (f (g args))))

(define (repeat n func)
  (define (iter n acc)
    (if (= n 1)
        (lambda args (apply acc args))
        (iter (- n 1) (lambda args
                        (cons (apply func args)
                              (apply acc args))))))
  (if (<= n 0) identity
      (iter n (lambda args (list (apply func args))))))
