#+begin_src racket
(define (make-account balance)
  (define (withdraw amount)
      (if (<= amount balance)
          (begin (set! balance (- balance amount))
                 balance)
          "Insufficient funds"))
  (define (deposit amount)
    (set! balance (+ balance amount))
    balance)
  (define (dispatch op)
    (cond [(eq? op 'withdraw) withdraw]
          [(eq? op 'deposit) deposit]
          [else (error "Unknown request -- MAKE-ACCOUNT" op)]))
  dispatch)

nil
(define acc (make-account 0))

(list
 ((acc 'deposit) 100)
 ((acc 'withdraw) 20)
 ((acc 'withdraw) 100))
#+end_src

#+RESULTS:
: (100 80 "Insufficient funds")

#+begin_src racket :results none
(define (lookup key table)
  (let ([record (assoc key (cdr table))])
    (if record
        (cdr record)
        #f)))

(define (insert! k v table)
  (let ([record (assoc k (cdr table))])
    (if record
        (set-cdr! record v)
        (set-cdr! table
                  (cons (cons k v) (cdr table))))

    (set-cdr! (list-tail table) (cons record nil))))

(define (make-table)
  (list '*table*))
#+end_src

** A Simulator of Digital Circuits

#+begin_src racket
(define (make-wire) ?)

(define (get-signal wire) ?)
(define (set-signal! wire v) ?)
(define (add-action! wire proc) ?)

(define (or-gate a b o) ?)
(define (not-gate in out)
  (define (invert-input)
    (let ([new-value (logical-not (get-signal in))])
      (after-delay not-gate-delay
                   (lambda ()
                     (set-signal! out new-value)))))
  (add-action! input invert-input)
  'ok)

(define (logical-not s)
  (cond [(= s 1) 0]
        [(= s 0) 1]
        [else (error "Invalid Signal" s)]))

(define (and-gate a b out)
  (define (and-action-procedure)
    (let ([new-value (logical-and (get-signal a) (get-signal b))])
      (after-delay and-gate-delay
                   (lambda ()
                     (set-signal! out new-value)))))
  (add-action! a and-action-procedure)
  (add-action! b and-action-procedure)
  'ok)

(define (half-adder a b s c)
  (let ([d (make-wire)] [e (make-wire)])
    (or-gate a b d)
    (and-gate a b c)
    (not-gate c e)
    (and-gate d e s)
    'ok))

(define (full-adder a b cin sum cout)
  (let ([s (make-wire)]
        [c1 (make-wire)]
        [c2 (make-wire)])
    (half-adder b cin s c1)
    (half-adder a s sum c2)
    (or-gate c1 c2 cout)
    'ok))
#+end_src
